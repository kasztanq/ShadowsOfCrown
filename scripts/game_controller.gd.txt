class_name GameController
extends Control

# --- ODWOŁANIA DO WĘZŁÓW UI (dopasowane do wcześniejszej hierarchii) ---
@onready var turn_label: Label = $SafeMarginContainer/MainVBox/TopHUD/TurnCounter
@onready var gold_label: Label = $SafeMarginContainer/MainVBox/TopHUD/GoldDisplay/GoldLabel

@onready var king_bar: ProgressBar = $SafeMarginContainer/MainVBox/MetersPanel/KingMeter/ProgressBar
@onready var church_bar: ProgressBar = $SafeMarginContainer/MainVBox/MetersPanel/ChurchMeter/ProgressBar
@onready var nobility_bar: ProgressBar = $SafeMarginContainer/MainVBox/MetersPanel/NobilityMeter/ProgressBar
@onready var suspicion_bar: ProgressBar = $SafeMarginContainer/MainVBox/MetersPanel/SuspicionMeter/ProgressBar

@onready var card_panel: NinePatchRect = $SafeMarginContainer/MainVBox/CardViewportArea/EventCard
@onready var speaker_name_label: Label = $SafeMarginContainer/MainVBox/CardViewportArea/EventCard/CardVBox/SpeakerName
@onready var speaker_portrait: TextureRect = $SafeMarginContainer/MainVBox/CardViewportArea/EventCard/CardVBox/SpeakerPortrait
@onready var description_label: RichTextLabel = $SafeMarginContainer/MainVBox/CardViewportArea/EventCard/CardVBox/EventDescription

@onready var choices_container: VBoxContainer = $SafeMarginContainer/MainVBox/ChoicesContainer

# --- STAN GRY ---
var current_turn: int = 1
var gold: int = 100
var stats: Dictionary = {
	"king_favor": 50.0,
	"church": 50.0,
	"nobility": 50.0,
	"suspicion": 15.0
}
var active_tokens: Array[String] = []

# Baza danych wczytana z JSON
var events_data: Array = []
var characters_data: Dictionary = {}
var current_event: Dictionary = {}
var is_processing_input: bool = false


func _ready() -> void:
	load_game_database()
	update_all_meters(false)
	load_next_event()


# --- ŁADOWANIE DANYCH JSON ---
func load_game_database() -> void:
	events_data = _load_json_file("res://data/events.json").get("events", [])
	var chars_array = _load_json_file("res://data/characters.json").get("characters", [])
	for character in chars_array:
		characters_data[character["id"]] = character


func _load_json_file(file_path: String) -> Dictionary:
	if not FileAccess.file_exists(file_path):
		push_warning("Brak pliku: " + file_path + " - wczytuję dane testowe.")
		return _get_fallback_mock_data()
	
	var file = FileAccess.open(file_path, FileAccess.READ)
	var content = file.get_as_text()
	var parsed = JSON.parse_string(content)
	return parsed if typeof(parsed) == TYPE_DICTIONARY else {}


# --- SELEKCJA I PREZENTACJA KART ---
func load_next_event() -> void:
	is_processing_input = true
	clear_choices()
	
	current_event = pick_valid_event()
	if current_event.is_empty():
		push_error("Brak dostępnych zdarzeń spełniających kryteria!")
		return
	
	# Płynne wejście nowej karty
	animate_card_entrance()
	populate_card_data(current_event)
	build_choice_buttons(current_event.get("choices", []))
	
	is_processing_input = false


func pick_valid_event() -> Dictionary:
	var available_events = []
	for ev in events_data:
		var conditions = ev.get("conditions", {})
		var turn_min = conditions.get("turn_min", 0)
		var suspicion_max = conditions.get("suspicion_max", 100)
		
		if current_turn >= turn_min and stats["suspicion"] <= suspicion_max:
			available_events.append(ev)
			
	if available_events.is_empty():
		return events_data[0] if not events_data.is_empty() else {}
	return available_events.pick_random()


func populate_card_data(ev: Dictionary) -> void:
	var speaker_id = ev.get("speaker_id", "")
	if characters_data.has(speaker_id):
		var char_info = characters_data[speaker_id]
		speaker_name_label.text = char_info.get("name", "Nieznany Dworzanin")
		var portrait_path = char_info.get("portrait_asset", "")
		if ResourceLoader.exists(portrait_path):
			speaker_portrait.texture = load(portrait_path)
	else:
		speaker_name_label.text = ev.get("title", "Audiencja")

	description_label.text = ev.get("text", "")


func build_choice_buttons(choices: Array) -> void:
	for choice in choices:
		var btn = Button.new()
		btn.custom_minimum_size = Vector2(0, 56)
		btn.text = choice.get("text", "Wybierz")
		btn.clip_text = false
		btn.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		
		# Sprawdzenie wymagań (np. posiadanie tokena/wywiadu)
		var reqs = choice.get("requirements", {})
		var can_afford = check_requirements(reqs)
		btn.disabled = not can_afford
		
		# Sygnały do obsługi wyboru i podglądu efektów
		var outcomes = choice.get("outcomes", {})
		btn.pressed.connect(_on_choice_selected.bind(outcomes))
		btn.mouse_entered.connect(_on_choice_hover_start.bind(outcomes))
		btn.mouse_exited.connect(_on_choice_hover_end)
		
		choices_container.add_child(btn)


func check_requirements(reqs: Dictionary) -> bool:
	if reqs.has("gold") and gold < reqs["gold"]:
		return false
	if reqs.has("intel") and not active_tokens.has("token_intel"):
		return false
	return true


# --- OBSŁUGA WYBORÓW I PĘTLA ROZGRYWKI ---
func _on_choice_selected(outcomes: Dictionary) -> void:
	if is_processing_input:
		return
	is_processing_input = true
	
	# Haptic feedback na Androidzie (50ms impuls)
	if OS.has_feature("mobile"):
		Input.vibrate_handheld(50)
	
	# Zastosowanie efektów
	apply_outcomes(outcomes)
	current_turn += 1
	turn_label.text = "Runda: %d" % current_turn
	gold_label.text = str(gold)
	
	# Płynne ukrycie obecnej karty i załadowanie nowej
	animate_card_exit()


func apply_outcomes(outcomes: Dictionary) -> void:
	gold += outcomes.get("gold", 0)
	stats["king_favor"] = clamp(stats["king_favor"] + outcomes.get("king_favor", 0), 0, 100)
	stats["church"] = clamp(stats["church"] + outcomes.get("church", 0), 0, 100)
	stats["nobility"] = clamp(stats["nobility"] + outcomes.get("nobility", 0), 0, 100)
	stats["suspicion"] = clamp(stats["suspicion"] + outcomes.get("suspicion", 0), 0, 100)
	
	if outcomes.has("add_token"):
		active_tokens.append(outcomes["add_token"])
		
	update_all_meters(true)
	check_game_over()


# --- SYSTEM WSKAŹNIKÓW I ANIMACJI TWEEN ---
func update_all_meters(animate: bool) -> void:
	var duration = 0.4 if animate else 0.0
	
	_animate_bar(king_bar, stats["king_favor"], duration)
	_animate_bar(church_bar, stats["church"], duration)
	_animate_bar(nobility_bar, stats["nobility"], duration)
	_animate_bar(suspicion_bar, stats["suspicion"], duration)


func _animate_bar(bar: ProgressBar, target_value: float, duration: float) -> void:
	if duration <= 0.0:
		bar.value = target_value
		return
	
	var tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	tween.tween_property(bar, "value", target_value, duration)


# Podświetlanie / zapowiedź wpływu na wskaźniki (Hover preview)
func _on_choice_hover_start(outcomes: Dictionary) -> void:
	_preview_pulse(king_bar, outcomes.get("king_favor", 0))
	_preview_pulse(church_bar, outcomes.get("church", 0))
	_preview_pulse(nobility_bar, outcomes.get("nobility", 0))
	_preview_pulse(suspicion_bar, outcomes.get("suspicion", 0))


func _on_choice_hover_end() -> void:
	# Przywrócenie standardowej skali i przezroczystości
	for bar in [king_bar, church_bar, nobility_bar, suspicion_bar]:
		var tween = create_tween()
		tween.tween_property(bar, "modulate", Color(1, 1, 1, 1), 0.15)


func _preview_pulse(bar: ProgressBar, delta: float) -> void:
	if delta == 0:
		return
	var color_highlight = Color(0.3, 1.0, 0.3) if delta > 0 else Color(1.0, 0.3, 0.3)
	var tween = create_tween().set_loops(2)
	tween.tween_property(bar, "modulate", color_highlight, 0.15)
	tween.tween_property(bar, "modulate", Color(1, 1, 1, 1), 0.15)


# --- ANIMACJE KARTY ---
func animate_card_entrance() -> void:
	card_panel.modulate.a = 0.0
	card_panel.position.y += 60.0
	
	var tween = create_tween().set_parallel(true).set_trans(Tween.TRANS_BACK).set_ease(Tween.EASE_OUT)
	tween.tween_property(card_panel, "modulate:a", 1.0, 0.35)
	tween.tween_property(card_panel, "position:y", card_panel.position.y - 60.0, 0.35)


func animate_card_exit() -> void:
	var tween = create_tween().set_parallel(true).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_IN)
	tween.tween_property(card_panel, "modulate:a", 0.0, 0.25)
	tween.tween_property(card_panel, "position:y", card_panel.position.y - 40.0, 0.25)
	
	# Po zakończeniu animacji usuwamy starą treść i wczytujemy kolejną
	tween.finished.connect(func():
		card_panel.position.y += 40.0 # Reset pozycji
		load_next_event()
	)


func clear_choices() -> void:
	for child in choices_container.get_children():
		child.queue_free()


# --- WARUNKI KOŃCA GRY ---
func check_game_over() -> void:
	if stats["suspicion"] >= 100.0:
		_trigger_game_over("Zostałeś zdemaskowany i wtrącony do lochu!")
	elif stats["king_favor"] <= 0.0:
		_trigger_game_over("Król uznał cię za bezużytecznego i skazał na wygnanie.")
	elif stats["church"] <= 0.0:
		_trigger_game_over("Święte Oficjum rzuciło na ciebie klątwę ekskomuniki.")
	elif stats["nobility"] <= 0.0:
		_trigger_game_over("Zbuntowani baronowie przeprowadzili zamach stanu.")


func _trigger_game_over(reason: String) -> void:
	description_label.text = "[color=red][b]KONIEC GRY[/b][/color]\n" + reason
	clear_choices()
	var restart_btn = Button.new()
	restart_btn.text = "Zacznij od nowa"
	restart_btn.pressed.connect(func(): get_tree().reload_current_scene())
	choices_container.add_child(restart_btn)
